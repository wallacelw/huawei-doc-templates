\section{Introduction}

\begin{objectives}
  \generalobjective{Validate the Huawei Cloud Data Platform deployment for the
    POC environment, ensuring that all core services (DataArts Studio, MRS,
    OBS, VPC) are provisioned correctly, accessible, and functioning as
    specified in the solution architecture.}

  \prerequisites
  \begin{itemize}
    \item Huawei Cloud account with IAM admin privileges in the \inlinecode{sa-brazil-1} region.
    \item VPC, subnet, and security groups already created by the infrastructure team.
    \item DataArts Studio instance (Dayu) provisioned and licensed.
    \item MRS cluster deployed with the required component stack (HDFS, YARN, Spark, Hive).
    \item OBS bucket created and accessible with the designated IAM policy.
    \item Test execution environment with network connectivity to the Huawei Cloud Console.
    \item \inlinecode{latexmk} and \inlinecode{terraform} CLI tools installed locally.
  \end{itemize}
\end{objectives}

\subsection{Test Scope}

\begin{table}[H]
  \begin{hutable}{|l|p{0.72\textwidth}|}
    \rowcolor{huaweired} \thd{Domain} & \thd{What Should Be Observed} \\
    \tbody
    Platform Architecture & Core services are deployed, healthy, and reachable via the Console API. \\
    Data Engineering & DataArts Studio pipelines can ingest, transform, and load data into the target storage. \\
    Data Storage & OBS buckets and MRS HDFS accept writes/reads with correct permissions. \\
    Security & IAM policies, VPC isolation, and security group rules enforce the intended access control. \\
  \end{hutable}
  \caption{Test scope by domain.}
\end{table}

\subsection{Preconditions and Preparations}

\begin{itemize}
  \item All infrastructure resources must be provisioned via Terraform and the \inlinecode{terraform apply} output must show zero errors.
  \item The MRS cluster must be in \textbf{Running} state before any data engineering test case is executed.
  \item Test data files (CSV/JSON) must be uploaded to the designated OBS bucket prior to pipeline execution.
  \item Each tester must have a personal IAM account with the \textbf{Dayu\_User} role assigned in DataArts Studio.
  \item Network firewall rules must allow inbound TCP 22 (SSH) and TCP 443 (HTTPS) from the test execution subnet.
\end{itemize}

\note{Ensure the test execution environment has the correct timezone set to \inlinecode{America/Sao\_Paulo} (GMT-3) so that log timestamps align with the Huawei Cloud Console.}

\begin{code}[bash]
# Verify CLI tools are installed
terraform version
obsutil version
kubectl version --client
\end{code}

\begin{tip}
Use \inlinecode{terraform plan} before \inlinecode{terraform apply} to preview
infrastructure changes and avoid unintended resource modifications.
\end{tip}

\subsection{Acceptance Method}

\begin{table}[H]
  \begin{hutable}{|l|p{0.72\textwidth}|}
    \rowcolor{huaweired} \thd{Result} & \thd{Explanation} \\
    \tbody
    Satisfied & The test case passes fully — observed behavior matches the expected result with no deviations. \\
    Satisfied but with reservations & The test case passes but minor deviations were observed that do not impact the POC objectives (documented in Remarks). \\
    Not satisfied & The test case fails — observed behavior does not match the expected result. A defect report must be filed. \\
    Untested & The test case could not be executed due to blocking dependencies or environment issues. \\
  \end{hutable}
  \caption{Acceptance criteria.}
\end{table}

\section{Test Cases}

\subsection{Platform Architecture}

\begin{testcase}{Verify MRS cluster deployment and health}
  \testobjective{Confirm that the MapReduce Service (MRS) cluster is deployed
    with the required components (HDFS, YARN, Spark, Hive) and all nodes
    report a healthy status.}
  \begin{testprerequisites}
    \teststep{Terraform infrastructure apply completed successfully.}
    \teststep{MRS cluster ID is available in the Terraform output.}
    \teststep{IAM user has \textbf{MRS\_Viewer} or higher role.}
  \end{testprerequisites}
  \begin{testprocedure}
    \teststep{Log in to the Huawei Cloud Console}
    \teststep{Navigate to \menu{Console, MapReduce Service, Clusters}}
    \teststep{Locate cluster \inlinecode{mrs-poc-cluster}}
    \teststep{Verify cluster status}
    \teststep{Review Component tab}
    \teststep{Review Node tab}
  \end{testprocedure}

  \image[width=0.8\linewidth]{common-assets/exemplo-menu.png}

  \begin{testexpected}
    \teststep{Cluster status is \textbf{Running}.}
    \teststep{All four components (HDFS, YARN, Spark, Hive) are listed with status \textbf{Normal}.}
    \teststep{All nodes report status \textbf{Running} with no alarms.}
  \end{testexpected}
  \testresult{\testresultbadge{Pass}}
  \testremarks{If any component shows \textbf{Abnormal}, check the MRS alarm page before marking the result.}
\end{testcase}

\begin{testcase}{Verify OBS bucket creation and accessibility}
  \testobjective{Confirm that the Object Storage Service (OBS) bucket used
    for the POC data lake is created, has the correct storage class, and
    is accessible with the designated IAM policy.}
  \begin{testprerequisites}
    \teststep{OBS bucket name is defined in the Terraform configuration.}
    \teststep{IAM policy granting read/write access to the bucket is attached to the test user.}
  \end{testprerequisites}
  \begin{testprocedure}
    \teststep{Log in to the Console}
    \teststep{Navigate to \menu{Console, Object Storage Service}}
    \teststep{Locate the bucket (e.g. \inlinecode{poc-datalake-raw})}
    \teststep{Verify the storage class is \textbf{Standard}}
    \teststep{Upload a test file (\inlinecode{test-upload.txt}) to the bucket}
    \teststep{Download the file and compare its content with the original}
    \teststep{Delete the test file}
  \end{testprocedure}
  \begin{testexpected}
    \teststep{Bucket exists with storage class \textbf{Standard}.}
    \teststep{File upload completes without error.}
    \teststep{Downloaded file content matches the original.}
    \teststep{File deletion succeeds.}
  \end{testexpected}
  \testresult{\testresultbadge{Pass}}
  \testremarks{OBS eventual consistency may cause a brief delay before a newly uploaded object appears in listings.}
\end{testcase}

\begin{testcase}{Verify VPC and security group configuration}
  \testobjective{Confirm that the VPC, subnet, and security groups are
    configured according to the network architecture diagram, with the
    correct CIDR blocks and inbound/outbound rules.}
  \begin{testprerequisites}
    \teststep{VPC and subnet IDs are available from the Terraform output.}
    \teststep{Security group names are documented in the solution architecture.}
  \end{testprerequisites}
  \begin{testprocedure}
    \teststep{Log in to the Console}
    \teststep{Navigate to \menu{Console, Virtual Private Cloud, VPCs}}
    \teststep{Locate the VPC (e.g. \inlinecode{vpc-poc}) and verify its CIDR block matches the architecture (e.g. \inlinecode{10.0.0.0/16})}
    \teststep{Click the subnet and verify its CIDR (e.g. \inlinecode{10.0.1.0/24})}
    \teststep{Navigate to \textbf{Security Groups} and verify inbound rules: TCP 22 from the bastion subnet, TCP 443 from the corporate proxy}
    \teststep{Verify outbound rules allow all traffic (default)}
  \end{testprocedure}
  \begin{testexpected}
    \teststep{VPC CIDR is \inlinecode{10.0.0.0/16}.}
    \teststep{Subnet CIDR is \inlinecode{10.0.1.0/24}.}
    \teststep{Inbound rules match the architecture: TCP 22 and TCP 443 from the specified sources.}
    \teststep{Outbound allows all traffic.}
  \end{testexpected}
  \testresult{\testresultbadge{Pass}}
  \testremarks{If the security group has additional rules beyond the architecture spec, document them in the Remarks field.}
\end{testcase}

\begin{testcase}{Verify data ingestion pipeline execution}
  \testobjective{Confirm that a DataArts Studio batch pipeline can
    successfully ingest CSV data from OBS, apply a basic transformation,
    and write the result to the target OBS path.}
  \begin{testprerequisites}
    \teststep{Test Case 4 completed (DataArts Studio workspace accessible).}
    \teststep{Source CSV file (\inlinecode{customers.csv}) exists in the OBS raw bucket.}
    \teststep{Target OBS path (\inlinecode{poc-datalake-curated/customers/}) is configured.}
    \teststep{Pipeline \inlinecode{pl-ingest-customers} is published in DataArts Studio.}
  \end{testprerequisites}
  \begin{testprocedure}
    \teststep{Navigate to Data Development in DataArts Studio}
    \teststep{Locate pipeline \inlinecode{pl-ingest-customers}}
    \teststep{Click Run to execute the pipeline}
    \teststep{Wait for pipeline status (timeout: 10 min)}
    \teststep{Verify output file in OBS}
    \teststep{Download and verify row count}
  \end{testprocedure}

  \begin{code}[bash]
# Verify the output file in OBS via CLI
obsutil ls obs://poc-datalake-curated/customers/ -limit=10
obsutil stat obs://poc-datalake-curated/customers/output.csv
  \end{code}

  \begin{testexpected}
    \teststep{Pipeline execution completes with status \textbf{Success}.}
    \teststep{Output file is created in the target OBS path.}
    \teststep{Row count of the output file matches the source file.}
  \end{testexpected}
  \testresult{\testresultbadge{Pass}}
  \testremarks{Pipeline execution time varies with data volume. For the POC, the source file contains approximately 10,000 rows.}
\end{testcase}

\begin{testcase}{Verify Spark SQL query on MRS cluster}
  \testobjective{Confirm that a Spark SQL query can be executed on the
    MRS cluster to read data from a Hive table and return correct
    results, validating the integration between MRS and the data lake.}
  \begin{testprerequisites}
    \teststep{Test Case 1 completed (MRS cluster healthy).}
    \teststep{Hive table \inlinecode{poc\_db.customers} exists and contains data loaded by the ingestion pipeline.}
    \teststep{SSH access to the MRS master node is configured.}
  \end{testprerequisites}
  \begin{testprocedure}
    \teststep{SSH into the MRS master node}
    \teststep{Run the Spark SQL CLI:}
\begin{code}[bash]
spark-sql --master yarn \
  --conf spark.sql.hive.convertMetastoreOrc=true \
  -e "SELECT COUNT(*) FROM poc_db.customers;"
\end{code}
    \teststep{Verify the returned count matches the expected row count (10,000)}
    \teststep{Run a sample query to verify data integrity:}
\begin{code}[bash]
spark-sql --master yarn \
  -e "SELECT customer_id, name FROM poc_db.customers LIMIT 5;"
\end{code}
    \teststep{Confirm the query returns 5 rows with non-null values}
  \end{testprocedure}

  \begin{testexpected}
    \teststep{\inlinecode{COUNT(*)} returns 10,000.}
    \teststep{The sample query returns 5 rows with valid \inlinecode{customer\_id} and \inlinecode{name} values.}
  \end{testexpected}
  \testresult{\testresultbadge{Pass}}
  \testremarks{If the Spark SQL CLI is not available, use the MRS Spark2x component's web UI to submit the query instead.}
\end{testcase}

\subsection{Security and Access Control}

\begin{testcase}{Verify IAM policy enforcement on OBS buckets}
  \testobjective{Confirm that IAM policies correctly restrict OBS bucket
    access — authorized users can read/write while unauthorized users are
    denied, and the bucket policy matches the principle of least privilege.}
  \begin{testprerequisites}
    \teststep{OBS bucket \inlinecode{poc-datalake-raw} exists with the designated IAM policy.}
    \teststep{Two test IAM users: one with \textbf{OBS\_ReadWrite} role, one with no OBS permissions.}
    \teststep{\inlinecode{obsutil} CLI configured for both users.}
  \end{testprerequisites}
  \begin{testprocedure}
    \teststep{Using the authorized user, upload a test file to \inlinecode{poc-datalake-raw}}
    \teststep{Using the authorized user, download and verify the file content}
    \teststep{Using the unauthorized user, attempt to upload a file — verify the request is denied (HTTP 403)}
    \teststep{Using the unauthorized user, attempt to list objects in the bucket — verify the request is denied}
    \teststep{Review the bucket policy in the Console at \menu{Console, Object Storage Service, Bucket Policies} and confirm it grants only the intended permissions}
  \end{testprocedure}

  \begin{code}[bash]
# Authorized user — upload and verify
obsutil cp test-file.txt obs://poc-datalake-raw/test-file.txt
obsutil ls obs://poc-datalake-raw/ -limit=5

# Unauthorized user — expect 403 errors
obsutil cp test-file.txt obs://poc-datalake-raw/unauthorized.txt
obsutil ls obs://poc-datalake-raw/
  \end{code}

  \begin{testexpected}
    \teststep{Authorized user can upload and download successfully.}
    \teststep{Unauthorized user receives HTTP 403 on both upload and list operations.}
    \teststep{Bucket policy grants only the intended read/write permissions to the designated roles.}
  \end{testexpected}
  \testresult{\testresultbadge{Pass}}
  \testremarks{Check the Cloud Trail logs if access denial is not returned as expected — there may be a policy inheritance from the project or domain level.}

  \note{IAM policy changes can take up to 5 minutes to propagate. Wait before
    re-testing after modifying a policy.}
\end{testcase}

\begin{testcase}{Verify DataArts Studio data masking rule}
  \testobjective{Confirm that sensitive data columns (e.g. PII fields) are
    masked according to the DataArts Studio data masking rule when queried
    by non-admin users.}
  \begin{testprerequisites}
    \teststep{Test Case 4 completed (DataArts Studio workspace accessible).}
    \teststep{Data masking rule configured for column \inlinecode{customers.ssn} (Social Security Number).}
    \teststep{Test user has \textbf{Dayu\_User} role (not admin).}
  \end{testprerequisites}
  \begin{testprocedure}
    \teststep{In the DataArts Studio console, navigate to \textbf{Data Architecture} $\rightarrow$ \textbf{Data Masking}}
    \teststep{Verify the masking rule for \inlinecode{customers.ssn} is active and uses the \textbf{Mask All} strategy}
    \teststep{Query the \inlinecode{customers} table as the test user via DataArts Studio SQL console}
    \teststep{Verify the \inlinecode{ssn} column returns masked values (e.g. \inlinecode{*********}) instead of real data}
    \teststep{Query the same table as an admin user and verify real values are returned}
  \end{testprocedure}
  \begin{testexpected}
    \teststep{Masking rule is active for \inlinecode{customers.ssn}.}
    \teststep{Non-admin user sees masked values in the \inlinecode{ssn} column.}
    \teststep{Admin user sees real values in the \inlinecode{ssn} column.}
  \end{testexpected}
  \testresult{\testresultbadge{Pass}}
  \testremarks{Data masking rules apply at query time — the underlying data in OBS/HDFS is not modified.}
\end{testcase}

\subsection{Environment Setup and Tooling}

\begin{testcase}{Verify Terraform infrastructure deployment}
  \testobjective{Confirm that the Terraform configuration deploys all
    required infrastructure resources (VPC, MRS, OBS, DataArts Studio)
    without errors and the state file reflects the expected resources.}
  \begin{testprerequisites}
    \teststep{\inlinecode{terraform} CLI installed (version $\geq$ 1.0).}
    \teststep{AK/SK credentials configured via environment variables or \param{provider.tf}.}
    \teststep{Terraform configuration files in the project directory.}
  \end{testprerequisites}
  \begin{testprocedure}
    \teststep{Navigate to the Terraform project directory}
    \teststep{Run \inlinecode{terraform init} to initialize the working directory}
    \teststep{Run \inlinecode{terraform plan} and review the planned changes — confirm all expected resources are listed}
    \teststep{Run \inlinecode{terraform apply -auto-approve} and wait for completion}
    \teststep{Run \inlinecode{terraform output} and verify all output values are populated}
    \teststep{Run \inlinecode{terraform state list} and confirm the resource count matches the architecture}
  \end{testprocedure}

  \begin{code}[bash]
# Initialize and deploy infrastructure
cd /opt/poc-infra/terraform
terraform init
terraform plan -out=tfplan
terraform apply tfplan

# Verify deployment
terraform output
terraform state list | wc -l
  \end{code}

  \begin{testexpected}
    \teststep{\inlinecode{terraform apply} completes with no errors.}
    \teststep{All output values are populated (cluster ID, bucket name, VPC ID).}
    \teststep{Resource count in state matches the architecture specification.}
  \end{testexpected}
  \testresult{\testresultbadge{Pass}}
  \testremarks{If \inlinecode{terraform apply} fails, check the AK/SK credentials and the \inlinecode{sa-brazil-1} region quota limits.}

  \begin{warning}
  Do not skip the \inlinecode{terraform plan} review step. Applying unreviewed
  changes to production-adjacent environments can cause data loss or service
  disruption.
  \end{warning}
\end{testcase}

\begin{testcase}{Verify test data upload to OBS}
  \testobjective{Confirm that all required test data files (CSV, JSON) can
    be uploaded to the designated OBS bucket and are accessible by the
    DataArts Studio pipeline.}
  \begin{testprerequisites}
    \teststep{Test Case 2 completed (OBS bucket accessible).}
    \teststep{Test data files prepared in the local directory \param{./test-data/}.}
    \teststep{\inlinecode{obsutil} CLI configured with the test user credentials.}
  \end{testprerequisites}
  \begin{testprocedure}
    \teststep{List the local test data files and verify their integrity (row count, schema)}
    \teststep{Upload all files to the OBS raw bucket using \inlinecode{obsutil}}
    \teststep{Verify each file exists in OBS by listing the bucket contents}
    \teststep{Download a sample file from OBS and compare it with the local original (checksum)}
    \teststep{In DataArts Studio, verify the data source connection can read the uploaded files}
  \end{testprocedure}

  \begin{code}[bash]
# Upload test data files to OBS
obsutil cp ./test-data/customers.csv obs://poc-datalake-raw/customers.csv
obsutil cp ./test-data/transactions.json obs://poc-datalake-raw/transactions.json

# Verify uploads
obsutil ls obs://poc-datalake-raw/ -limit=20
  \end{code}

  \begin{testexpected}
    \teststep{All test data files are uploaded without errors.}
    \teststep{Bucket listing shows all uploaded files with correct sizes.}
    \teststep{Downloaded file checksum matches the local original.}
    \teststep{DataArts Studio data source connection can read the files.}
  \end{testexpected}
  \testresult{\testresultbadge{Pass}}
  \testremarks{Large files ($>$1 GB) should be uploaded using the OBS multipart upload feature for better reliability.}

  \note{Refer to the \weblink{https://support.huaweicloud.com/intl/en-us/obs/index.html}{Huawei Cloud OBS Documentation} for advanced upload options and troubleshooting.}
\end{testcase}
